cmake_minimum_required(VERSION 3.16)
project(boot_swap_lab LANGUAGES CXX)
enable_testing()
add_executable(boot_swap_test swap_test.cpp)
target_compile_features(boot_swap_test PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(boot_swap_test PRIVATE /W4 /WX)
else()
    target_compile_options(boot_swap_test PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME boot_swap_power_loss COMMAND boot_swap_test)
