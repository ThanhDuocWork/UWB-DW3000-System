cmake_minimum_required(VERSION 3.16)

set(SWAP_BOARD_ROLE "anchor" CACHE STRING "anchor or tag")
set(SWAP_VARIANT "A" CACHE STRING "A or B")
if(NOT SWAP_BOARD_ROLE MATCHES "^(anchor|tag)$")
    message(FATAL_ERROR "SWAP_BOARD_ROLE must be anchor or tag")
endif()
if(NOT SWAP_VARIANT MATCHES "^(A|B)$")
    message(FATAL_ERROR "SWAP_VARIANT must be A or B")
endif()

set(PROJECT_VER "${SWAP_VARIANT}.1")
if(SWAP_BOARD_ROLE STREQUAL "anchor" AND SWAP_VARIANT STREQUAL "B")
    set(components_root "${CMAKE_CURRENT_LIST_DIR}/../../components")
    foreach(component board utils platform_hal drivers/dw3000 uwb ranging)
        list(APPEND EXTRA_COMPONENT_DIRS "${components_root}/${component}")
    endforeach()
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(uwb_swap_${SWAP_BOARD_ROLE})
if(SWAP_BOARD_ROLE STREQUAL "anchor" AND SWAP_VARIANT STREQUAL "B")
    idf_component_get_property(board_lib board COMPONENT_LIB)
    target_compile_definitions(${board_lib} PRIVATE CONFIG_UWB_ROLE_ANCHOR=1 CONFIG_UWB_NODE_ID=100)
endif()
